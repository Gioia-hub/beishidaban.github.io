<script>

// ======================== 北师大版 模拟数据（请替换为真实数据） ========================

const baseData = {

    // 年级 -> 册次 -> 知识点列表

    "1": {

        "上": [

            { id: "1-上-1", name: "数一数", video: "https://www.example.com/video1.mp4", questions: [{ q: "3+2=?", options: ["4","5","6"], answer: 1 }] },

            { id: "1-上-2", name: "比一比", video: "https://www.example.com/video2.mp4", questions: [{ q: "5比3多几？", options: ["1","2","3"], answer: 1 }] }

        ],

        "下": [

            { id: "1-下-1", name: "加与减（一）", video: "https://www.example.com/video3.mp4", questions: [{ q: "8-3=?", options: ["4","5","6"], answer: 1 }] }

        ]

    },

    "2": {

        "上": [

            { id: "2-上-1", name: "长度单位", video: "https://www.example.com/video4.mp4", questions: [{ q: "1米=？厘米", options: ["10","100","1000"], answer: 1 }] }

        ],

        "下": [

            { id: "2-下-1", name: "有余数的除法", video: "https://www.example.com/video5.mp4", questions: [{ q: "7÷3=？余几？", options: ["1","2","3"], answer: 1 }] }

        ]

    },

    // 其他年级类似... 此处省略，实际使用时请补全所有年级和知识点

    "3": { "上": [], "下": [] },

    "4": { "上": [], "下": [] },

    "5": { "上": [], "下": [] },

    "6": { "上": [], "下": [] }

};


// 补充完整（为了让演示更丰富，为3-6年级也加几个占位知识点）

for (let g=3; g<=6; g++) {

    baseData[g] = {

        "上": [

            { id: `${g}-上-1`, name: `第${g}年级上册知识点1`, video: "https://www.example.com/video.mp4", questions: [{ q: "示例题？", options: ["A","B","C"], answer: 0 }] },

            { id: `${g}-上-2`, name: `第${g}年级上册知识点2`, video: "https://www.example.com/video.mp4", questions: [{ q: "示例题？", options: ["A","B","C"], answer: 1 }] }

        ],

        "下": [

            { id: `${g}-下-1`, name: `第${g}年级下册知识点1`, video: "https://www.example.com/video.mp4", questions: [{ q: "示例题？", options: ["A","B","C"], answer: 2 }] }

        ]

    };

}


// ======================== 应用状态 ========================

let currentGrade = '1';

let currentSemester = '上';

let currentKnowledge = null;       // 当前选中的知识点对象

let currentQuestionIndex = 0;

let wrongQuestions = JSON.parse(localStorage.getItem('wrongQuestions')) || [];


// ======================== DOM引用 ========================

const gradeSelector = document.getElementById('gradeSelector');

const semesterTabs = document.getElementById('semesterTabs');

const knowledgeList = document.getElementById('knowledgeList');

const videoContainer = document.getElementById('videoContainer');

const videoIframe = document.getElementById('videoIframe');

const exerciseArea = document.getElementById('exerciseArea');

const questionDisplay = document.getElementById('questionDisplay');

const optionsDisplay = document.getElementById('optionsDisplay');

const submitBtn = document.getElementById('submitAnswer');

const nextBtn = document.getElementById('nextQuestion');

const feedback = document.getElementById('feedback');

const wrongList = document.getElementById('wrongList');

const clearWrongBtn = document.getElementById('clearWrong');

const toast = document.getElementById('toast');


// ======================== 渲染函数 ========================

function renderGradeButtons() {

    gradeSelector.innerHTML = '';

    for (let i=1; i<=6; i++) {

        const btn = document.createElement('button');

        btn.className = `grade-btn ${i == currentGrade ? 'active' : ''}`;

        btn.textContent = `一年级${i>1?i:''}`;

        btn.dataset.grade = i;

        btn.addEventListener('click', () => {

            currentGrade = i;

            renderGradeButtons();

            renderKnowledge();

            hideVideoAndExercise();

        });

        gradeSelector.appendChild(btn);

    }

}


function renderKnowledge() {

    const data = baseData[currentGrade]?.[currentSemester] || [];

    knowledgeList.innerHTML = '';

    data.forEach(item => {

        const div = document.createElement('div');

        div.className = 'knowledge-item';

        div.textContent = item.name;

        div.addEventListener('click', () => {

            currentKnowledge = item;

            currentQuestionIndex = 0;

            showVideo(item.video);

            showExercise(item.questions);

            renderWrongBook();

        });

        knowledgeList.appendChild(div);

    });

    // 如果当前没有选中知识点，清空视频和练习

    if (data.length === 0) {

        hideVideoAndExercise();

    }

}


function showVideo(url) {

    videoContainer.style.display = 'block';

    videoIframe.src = url; // 实际使用时，可能是视频地址，这里直接赋值

    // 如果是普通mp4，可用video标签，但为了演示，我们用iframe，假设是网页视频或可嵌入链接

    // 为了支持mp4，可改用<video>，这里简单处理

}


function hideVideoAndExercise() {

    videoContainer.style.display = 'none';

    exerciseArea.style.display = 'none';

    videoIframe.src = '';

}


function showExercise(questions) {

    if (!questions || questions.length === 0) {

        exerciseArea.style.display = 'none';

        return;

    }

    exerciseArea.style.display = 'block';

    renderQuestion(questions, currentQuestionIndex);

}


function renderQuestion(questions, index) {

    if (index >= questions.length) {

        questionDisplay.textContent = '🎉 恭喜完成所有题目！';

        optionsDisplay.innerHTML = '';

        feedback.textContent = '';

        submitBtn.style.display = 'none';

        nextBtn.style.display = 'none';

        return;

    }

    const q = questions[index];

    questionDisplay.textContent = `${index+1}. ${q.q}`;

    optionsDisplay.innerHTML = '';

    q.options.forEach((opt, i) => {

        const label = document.createElement('label');

        label.innerHTML = `<input type="radio" name="option" value="${i}"> ${opt}`;

        optionsDisplay.appendChild(label);

    });

    feedback.textContent = '';

    submitBtn.style.display = 'inline-block';

    nextBtn.style.display = 'inline-block';

    // 重置选项

    document.querySelectorAll('input[name="option"]').forEach(el => el.checked = false);

}


// ======================== 练习逻辑 ========================

submitBtn.addEventListener('click', () => {

    const selected = document.querySelector('input[name="option"]:checked');

    if (!selected) {

        showToast('请先选择一个选项');

        return;

    }

    const answer = parseInt(selected.value);

    const q = currentKnowledge.questions[currentQuestionIndex];

    if (answer === q.answer) {

        feedback.textContent = '✅ 回答正确！';

        feedback.style.color = '#38a169';

    } else {

        feedback.textContent = '❌ 错误，已加入错题本';

        feedback.style.color = '#e53e3e';

        // 保存错题

        const wrong = {

            grade: currentGrade,

            semester: currentSemester,

            knowledge: currentKnowledge.name,

            question: q.q,

            correctAnswer: q.options[q.answer],

            yourAnswer: q.options[answer]

        };

        wrongQuestions.push(wrong);

        localStorage.setItem('wrongQuestions', JSON.stringify(wrongQuestions));

        renderWrongBook();

    }

});


nextBtn.addEventListener('click', () => {

    const questions = currentKnowledge.questions;

    currentQuestionIndex++;

    if (currentQuestionIndex < questions.length) {

        renderQuestion(questions, currentQuestionIndex);

    } else {

        renderQuestion(questions, questions.length); // 显示完成

    }

});


// ======================== 错题本 ========================

function renderWrongBook() {

    if (wrongQuestions.length === 0) {

        wrongList.innerHTML = '<p style="color:#718096;">暂无错题，继续加油！</p>';

        return;

    }

    wrongList.innerHTML = '';

    wrongQuestions.slice().reverse().forEach((item, idx) => {

        const div = document.createElement('div');

        div.className = 'wrong-item';

        div.innerHTML = `

            <strong>${item.grade}年级${item.semester}册 · ${item.knowledge}</strong><br>

            ${item.question}<br>

            你的答案：${item.yourAnswer} ｜ 正确答案：${item.correctAnswer}

        `;

        wrongList.appendChild(div);

    });

}


clearWrongBtn.addEventListener('click', () => {

    if (confirm('确定清空所有错题吗？')) {

        wrongQuestions = [];

        localStorage.setItem('wrongQuestions', JSON.stringify(wrongQuestions));

        renderWrongBook();

        showToast('错题已清空');

    }

});


// ======================== 册次切换 ========================

semesterTabs.addEventListener('click', (e) => {

    if (e.target.classList.contains('semester-tab')) {

        document.querySelectorAll('.semester-tab').forEach(t => t.classList.remove('active'));

        e.target.classList.add('active');

        currentSemester = e.target.dataset.semester;

        renderKnowledge();

        hideVideoAndExercise();

    }

});


// ======================== Toast ========================

function showToast(msg) {

    toast.textContent = msg;

    toast.classList.add('show');

    setTimeout(() => toast.classList.remove('show'), 2000);

}


// ======================== 初始化 ========================

renderGradeButtons();

renderKnowledge();

renderWrongBook();

hideVideoAndExercise();


// ======================== 扩展：自动播放视频（占位） ========================

// 如果视频链接是mp4，可改为video标签，这里为了演示，直接使用iframe加载任意链接

</script>
